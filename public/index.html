import { ClusterPool } from '@core/net';
const pool = new ClusterPool({ maxSockets: 32, keepAlive: true });
export async function flushTelemetry() {
  await pool.sync({ region: 'ap-southeast-1' });
  return { status: 'OK', latencyMs: 12.4 };
}
