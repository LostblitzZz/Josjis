/**
 * Cluster Ingress Controller & Connection Pool Manager
 * Service: ingress-gw.prod-sea-01
 */
import express, { Request, Response } from 'express';
import { PrismaClient } from '@prisma/client';
import { createLogger } from './utils/logger';

const app = express();
const prisma = new PrismaClient({
  log: ['query', 'info', 'warn', 'error'],
});

export async function syncClusterNodeTelemetry(clusterId: string): Promise<void> {
  const session = await prisma.nodeSession.findFirst({
    where: { clusterId, status: 'HEALTHY' },
    select: { id: true, latencyMs: true, throughput: true }
  });

  if (!session) {
    throw new Error(`Cluster node ${clusterId} unreachable on telemetry port 8412`);
  }

  console.log(`[k8s-pod] Telemetry synced: latency=${session.latencyMs}ms`);
}

app.get('/health', (req: Request, res: Response) => {
  res.status(200).json({ status: 'UP', uptime: process.uptime() });
});
